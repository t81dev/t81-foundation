cmake_minimum_required(VERSION 3.20)
project(arch003 LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "ARCH-002-R1 requires a Clang-family compiler")
endif()
set(ARCH003_PROFILE D CACHE STRING "D, R, or S")
option(ARCH003_COMPONENT_TESTS "Register bounded component checks" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
# Clear generator defaults; profiles below are the complete study flags.
set(CMAKE_CXX_FLAGS_DEBUG "")
set(CMAKE_CXX_FLAGS_RELEASE "")
add_compile_options(-Wall -Wextra -Werror -fno-lto)
add_link_options(-fno-lto)
if(ARCH003_PROFILE STREQUAL "D")
  add_compile_options(-O0 -g -UNDEBUG)
elseif(ARCH003_PROFILE STREQUAL "R")
  add_compile_options(-O3 -DNDEBUG)
elseif(ARCH003_PROFILE STREQUAL "S")
  add_compile_options(-O1 -g -DNDEBUG -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
else()
  message(FATAL_ERROR "Unknown profile")
endif()
include_directories(include)
add_library(arch003_support STATIC src/schedule.cpp src/sha256.cpp src/wire.cpp)
add_library(arch003_candidates STATIC src/adapters.cpp src/admission.cpp src/count_tc2.cpp src/count_byte.cpp src/count_planes.cpp src/count_bc2.cpp src/count_pt5.cpp src/count_native.cpp)
add_library(arch003_sink STATIC src/sink.cpp)
foreach(kind scalar histogram)
  add_executable(arch003_${kind}_oracle src/oracle_${kind}.cpp src/oracle_main.cpp)
  target_link_libraries(arch003_${kind}_oracle PRIVATE arch003_support)
endforeach()
add_executable(arch003_worker src/worker.cpp)
target_link_libraries(arch003_worker PRIVATE arch003_candidates arch003_sink arch003_support)
add_executable(arch003_component_checks tests/component_checks.cpp)
target_link_libraries(arch003_component_checks PRIVATE arch003_candidates arch003_sink arch003_support)
# Oracle component driver has no candidate linkage; oracle TUs stay separate binaries.
if(ARCH003_PROFILE STREQUAL "S")
  add_executable(arch003_sanitizer_probe tests/sanitizer_probe.cpp)
endif()
if(ARCH003_COMPONENT_TESTS)
  enable_testing()
  add_test(NAME components COMMAND arch003_component_checks)
  add_test(NAME scalar_components COMMAND arch003_scalar_oracle --component)
  add_test(NAME histogram_components COMMAND arch003_histogram_oracle --component)
endif()
