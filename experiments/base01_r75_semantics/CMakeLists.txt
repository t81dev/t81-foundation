cmake_minimum_required(VERSION 3.20)
project(base01_r75_semantics LANGUAGES CXX)
if(NOT DEFINED AUDITED_SOURCE OR NOT DEFINED AUDITED_BUILD)
  message(FATAL_ERROR "Supply separate pinned AUDITED_SOURCE and correctness-only AUDITED_BUILD")
endif()
execute_process(COMMAND git -C "${AUDITED_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE audited_head OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT audited_head STREQUAL "cda9135d36144d4de2771af174e52ac7b30a83c9")
  message(FATAL_ERROR "Wrong audited source revision")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
add_executable(r75_correctness probe.cpp)
target_include_directories(r75_correctness PRIVATE "${AUDITED_SOURCE}/include" "${AUDITED_SOURCE}/tests/cpp")
set(archives)
foreach(lib vm axion isa experimental ffi core)
  if(NOT EXISTS "${AUDITED_BUILD}/libt81_${lib}.a")
    message(FATAL_ERROR "Build audited correctness targets first: missing ${lib}")
  endif()
  list(APPEND archives "${AUDITED_BUILD}/libt81_${lib}.a")
endforeach()
target_link_libraries(r75_correctness PRIVATE ${archives} ${archives})
if(UNIX AND NOT APPLE)
  target_link_libraries(r75_correctness PRIVATE pthread dl)
endif()
